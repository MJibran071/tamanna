    80→function getAgentIconKey(agent: string): string {
    81→  const key = agent.toLowerCase();
    82→  const map: Record<string, string> = {
    83→    researcher: 'search',
    84→    writer: 'filetext',
    85→    coder: 'code2',
    86→    browser: 'globe',
    87→    analyst: 'calculator',
    88→    planner: 'brain',
    89→    executor: 'zap',
    90→    vision: 'eye',
    91→    translator: 'languages',
    92→    summarizer: 'filetext',
    93→    math: 'calculator',
    94→    search: 'search',
    95→  };
    96→  for (const [k, v] of Object.entries(map)) {
    97→    if (key.includes(k)) return v;
    98→  }
    99→  return 'bot';
   100→}

   102→function DynamicAgentIcon({ agent, className, size = 'sm' }: { agent: string; className?: string; size?: 'sm' | 'md' }) {
   103→  const iconKey = getAgentIconKey(agent);
   104→  const cls = className ?? '';
   105→  const IconComponent = size === 'sm' ? 'w-3.5 h-3.5' : 'w-4 h-4';
   106→  switch (iconKey) {
   107→    case 'search':
   108→      return <Search className={`${cls} ${IconComponent}`} />;
   109→    case 'filetext':
   110→      return <FileText className={`${cls} ${IconComponent}`} />;
   111→    case 'code2':
   112→      return <Code2 className={`${cls} ${IconComponent}`} />;
   113→    case 'globe':
   114→      return <Globe className={`${cls} ${IconComponent}`} />;
   115→    case 'calculator':
   116→      return <Calculator className={`${cls} ${IconComponent}`} />;
   117→    case 'brain':
   118→      return <Brain className={`${cls} ${IconComponent}`} />;
   119→    case 'zap':
   120→      return <Zap className={`${cls} ${IconComponent}`} />;
   121→    default:
   122→      return <Bot className={`${cls} ${IconComponent}`} />;
   123→  }
   124→}

   126→function formatDuration(ms: number): string {
   127→  if (ms < 1000) return `${ms}ms`;
   128→  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
   129→  return `${Math.floor(ms / 60000)}m ${Math.floor((ms % 60000) / 1000)}s`;
   130→}

   140→    default:
   141→      return 'outline';
   142→  }

   212→function StepTimeline({ steps }: { steps: TaskStep[] }) {
   213→  const completedCount = steps.filter(s => s.status === 'completed').length;
   214→  const progressPercent = steps.length > 0 ? (completedCount / steps.length) * 100 : 0;
   215→
   216→  return (
   217→    <div className="relative">
   218→      {/* Progress bar */}
   219→      {steps.length > 1 && (
   220→        <div className="absolute left-[15px] top-0 bottom-0 w-[2px]">
   221→          <motion.div
   222→            className="w-full bg-gradient-to-b from-lumina-primary/40 to-lumina-primary/10 rounded-full"
   223→            initial={{ height: '0%' }}
   224→            animate={{ height: `${progressPercent}%` }}
   225→            transition={{ duration: 0.6, ease: 'easeOut' }}
   226→          />
   227→        </div>
   228→      )}
   229→
   230→      <div className="space-y-2">
   231→        {steps.map((step, idx) => (
   232→          <StepTimelineItem key={step.id} step={step} index={idx} />
   233→        ))}
   234→      </div>
   235→    </div>
   236→  );

   712→// ─── Main Dashboard ─────────────────────────────────────────────────────────
   713→
   714→export default function TaskDashboard() {
   715→  const currentPlan = useAgentStore((s) => s.currentPlan);
   716→  const stepStates = useAgentStore((s) => s.stepStates);
   717→  const agentState = useAgentStore((s) => s.agentState);
   718→  const messages = useAgentStore((s) => s.messages);
   719→  const [expandedTask, setExpandedTask] = useState<string | null>(null);
   720→
   721→  // Build the "active" task from current Zustand state (real-time execution)
   722→  const activeTask: TaskExecution | null = useMemo(() => {
   723→    if (stepStates.length === 0) return null;
   724→
   725→    return {
   726→      id: 'active',
   727→      prompt: messages.find(m => m.role === 'user')?.content || 'Current Task',
   728→      status: agentState === 'executing' || agentState === 'planning' ? 'running'
   729→        : stepStates.every(s => s.status === 'completed') ? 'completed'
   730→        : stepStates.some(s => s.status === 'failed') ? 'failed'
   731→        : 'pending',
   732→      steps: stepStates.map(s => ({
   733→        id: `step-${s.stepIndex}`,
   734→        agent: s.agent,
   735→        description: s.description,
   736→        status: s.status,
   737→        result: s.result,
   738→        error: s.error,
   739→      })),
   740→      startedAt: new Date().toISOString(),
   741→      completedAt: stepStates.every(s => s.status !== 'pending' && s.status !== 'running')
   742→        ? new Date().toISOString()
   743→        : null,
   744→      artifacts: [],
   745→    };
   746→  }, [stepStates, agentState, messages]);
   747→
   748→  // Build historical "completed" tasks from message history
   749→  const historicalTasks: TaskExecution[] = useMemo(() => {
   750→    // Extract task-like patterns from past messages
   751→    const tasks: TaskExecution[] = [];
   752→    // For now, we show the active task prominently
   753→    // Historical tasks would come from the database in future iterations
   754→    return tasks;
   755→  }, []);
   756→
   757→  // Calculate stats
   758→  const stats: TaskStats = useMemo(() => {
   759→    const allTasks = [...historicalTasks];
   760→    if (activeTask) allTasks.push(activeTask);
   761→
   762→    const completed = allTasks.filter(t => t.status === 'completed');
   763→    const failed = allTasks.filter(t => t.status === 'failed');
   764→    const withDuration = allTasks.filter(t => t.durationMs);
   765→
   766→    return {
   767→      totalTasks: allTasks.length,
   768→      completedTasks: completed.length,
   769→      failedTasks: failed.length,
   770→      avgDurationMs: withDuration.length > 0
   771→        ? Math.round(withDuration.reduce((sum, t) => sum + (t.durationMs || 0), 0) / withDuration.length)
   772→        : 0,
   773→      successRate: allTasks.length > 0
   774→        ? Math.round((completed.length / allTasks.length) * 100)
   775→        : 0,
   776→      totalSteps: allTasks.reduce((sum, t) => sum + t.steps.length, 0),
   777→    };
   778→  }, [activeTask, historicalTasks]);
   779→
   780→  const isCurrentlyExecuting = agentState === 'executing' || agentState === 'planning';
   781→
   782→  return (
   783→    <div className="w-full space-y-4">
   784→      {/* Header */}
   785→      <div className="flex items-center justify-between">
   786→        <div className="flex items-center gap-2">
   787→          <div className="p-1.5 rounded-xl bg-lumina-primary/10 border border-lumina-primary/20">
   788→            <BrainCircuit className="w-4 h-4 text-lumina-primary" />
   789→          </div>
   790→          <div>
   791→            <h2 className="font-[family-name:var(--font-display)] text-lg font-semibold text-lumina-on-surface">
   792→              Task Dashboard
   793→            </h2>
   794→            <p className="text-[11px] text-lumina-on-surface-variant/50">
   795→              Real-time agent execution monitoring
   796→            </p>
   797→          </div>
   798→        </div>
   799→        {isCurrentlyExecuting && (
   800→          <motion.span
   801→            initial={{ opacity: 0, scale: 0.8 }}
   802→            animate={{ opacity: 1, scale: 1 }}
   803→            className="flex items-center gap-1.5 px-2.5 py-1 rounded-full bg-lumina-primary/10 border border-lumina-primary/20 text-[11px] font-medium text-lumina-primary"
   804→          >
   805→            <motion.span
   806→              className="w-2 h-2 rounded-full bg-lumina-primary"
   807→              animate={{ scale: [1, 1.4, 1], opacity: [1, 0.7, 1] }}
   808→              transition={{ duration: 1.5, repeat: Infinity }}
   809→            />
   810→            Live
   811→          </motion.span>
   812→        )}
   813→      </div>
   814→
   815→      {/* Stats */}
   816→      {stats.totalTasks > 0 && <StatsRow stats={stats} />}
   817→
   818→      {/* Active Task Hero (during execution) */}
   819→      <AnimatePresence>
   820→        {activeTask && isCurrentlyExecuting && (
   821→          <ActiveTaskHero task={activeTask} />
   822→        )}
   823→      </AnimatePresence>
   824→
   825→      {/* Task List */}
   826→      <div className="space-y-3">
   827→        {/* Active task card (when not executing, show completed/failed state) */}
   828→        {activeTask && !isCurrentlyExecuting && stepStates.length > 0 && (
   829→          <TaskCard
   830→            key={activeTask.id}
   831→            task={activeTask}
   832→            isExpanded={expandedTask === activeTask.id}
   833→            onToggle={() => setExpandedTask(prev => prev === activeTask.id ? null : activeTask.id)}
   834→          />
   835→        )}
   836→
   837→        {/* Historical tasks */}
   838→        {historicalTasks.map(task => (
   839→          <TaskCard
   840→            key={task.id}
   841→            task={task}
   842→            isExpanded={expandedTask === task.id}
   843→            onToggle={() => setExpandedTask(prev => prev === task.id ? null : task.id)}
   844→          />
   845→        ))}
   846→
   847→        {/* Empty state */}
   848→        {!activeTask && historicalTasks.length === 0 && <EmptyState />}
   849→      </div>
   850→    </div>
   851→  );
   852→}
