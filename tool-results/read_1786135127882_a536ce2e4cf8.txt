    64→/* ─── helpers ────────────────────────────────────────────── */
    65→function loadProfile(): UserProfile {
    66→  try {
    67→    const raw = localStorage.getItem(PROFILE_KEY);
    68→    if (raw) {
    69→      return { ...DEFAULT_PROFILE, ...JSON.parse(raw) };
    70→    }
    71→  } catch {
    72→    /* ignore */
    73→  }
    74→  return DEFAULT_PROFILE;
    75→}
    76→
    77→function saveProfile(profile: UserProfile) {
    78→  try {
    79→    localStorage.setItem(PROFILE_KEY, JSON.stringify(profile));
    80→  } catch {
    81→    /* storage full */
    82→  }
    83→}
    84→
    85→function getInitials(name: string): string {
    86→  if (!name || !name.trim()) return "U";
    87→  const parts = name.trim().split(/\s+/);
    88→  if (parts.length >= 2) {
    89→    return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
    90→  }
    91→  return parts[0][0].toUpperCase();
    92→}
    93→
    94→/** Simple deterministic hash from a string to a number 0‒1 */
    95→function hashStr(str: string): number {
    96→  let h = 0;
    97→  for (let i = 0; i < str.length; i++) {
    98→    h = (h * 31 + str.charCodeAt(i)) | 0;
    99→  }
   100→  return ((h & 0x7fffffff) % 1000) / 1000;
   101→}

   103→/** Format a Date to "Mon DD" like "Jun 15" */
   104→function shortDay(d: Date): string {
   105→  return d.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   106→}
   107→
   108→/** ISO date string "YYYY-MM-DD" */
   109→function isoDate(d: Date): string {
   110→  return d.toISOString().slice(0, 10);
   111→}

   113→/* ─── activity tracking helpers ─────────────────────────────── */
   114→
   115→/** Track today as an active day (called once per render when sheet is open) */
   116→function recordTodayActivity() {
   117→  const today = isoDate(new Date());
   118→  const log = getActivityLog();
   119→  if (log[today]) return; // already recorded today
   120→  log[today] = true;
   121→  localStorage.setItem(ACTIVITY_KEY, JSON.stringify(log));
   122→}
   123→
   124→/** Load the full activity log { "YYYY-MM-DD": true } */
   125→function getActivityLog(): Record<string, boolean> {
   126→  try {
   127→    const raw = localStorage.getItem(ACTIVITY_KEY);
   128→    if (raw) return JSON.parse(raw);
   129→  } catch {
   130→    /* ignore */
   131→  }
   132→  return {};
   133→}
   134→
   135→/** Seed historical activity from the first-use / onboarding date */
   136→function ensureFirstUseDate(): string | null {
   137→  // Prefer the explicit key
   138→  let first = localStorage.getItem(FIRST_USE_KEY);
   139→  if (!first) {
   140→    first = localStorage.getItem(ONBOARDING_KEY) === "1" ? "seeded" : null;
   141→  }
   142→  if (!first) {
   143→    // Check profile key timestamp as heuristic
   144→    try {
   145→      const profileRaw = localStorage.getItem(PROFILE_KEY);
   146→      if (profileRaw) {
   147→        first = "seeded";
   148→      }
   149→    } catch {
   150→      /* ignore */
   151→    }
   152→  }
   153→  if (!first) return null;
   154→  if (first === "seeded") {
   155→    // Set today as first use if not already set
   156→    localStorage.setItem(FIRST_USE_KEY, isoDate(new Date()));
   157→    return localStorage.getItem(FIRST_USE_KEY);
   158→  }
   159→  return first;
   160→}
   161→
   162→/** Calculate the current streak (consecutive active days ending today/yesterday) */
   163→function calculateStreak(): number {
   164→  const log = getActivityLog();
   165→  const today = new Date();
   166→  let streak = 0;
   167→  for (let i = 0; i < 365; i++) {
   168→    const d = new Date(today);
   169→    d.setDate(d.getDate() - i);
   170→    const key = isoDate(d);
   171→    if (log[key]) {
   172→      streak++;
   173→    } else {
   174→      // Allow today to be not yet recorded — if yesterday is active, streak is at least 1
   175→      if (i === 0) continue;
   176→      break;
   177→    }
   178→  }
   179→  return streak;
   180→}
   181→
   182→/** Count total distinct active days */
   183→function countActiveDays(): number {
   184→  return Object.keys(getActivityLog()).length;
   185→}
   186→
   187→/** Detect browser name from UA */
   188→function detectBrowser(): string {
   189→  if (typeof navigator === "undefined") return "Unknown";
   190→  const ua = navigator.userAgent;
   191→  if (ua.includes("Firefox")) return "Firefox";
   192→  if (ua.includes("Edg")) return "Edge";
   193→  if (ua.includes("Chrome")) return "Chrome";
   194→  if (ua.includes("Safari")) return "Safari";
   195→  return "Other";
   196→}
