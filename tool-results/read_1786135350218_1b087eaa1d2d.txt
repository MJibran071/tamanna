   177→
   178→function loadSettings(): TamannaSettings {
   179→  try {
   180→    const raw = localStorage.getItem(SETTINGS_KEY);
   181→    if (raw) {
   182→      return { ...DEFAULT_SETTINGS, ...JSON.parse(raw) };
   183→    }
   184→  } catch {
   185→    /* ignore parse errors */
   186→  }
   187→  return DEFAULT_SETTINGS;
   188→}
   189→
   190→function saveSettings(settings: TamannaSettings) {
   191→  try {
   192→    localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings));
   193→  } catch {
   194→    /* storage full */
   195→  }
   196→}

   275→      const res = await fetch("/api/scheduled-tasks");

   339→      toast({ title: "Error", description: "Failed to create task", variant: "destructive" });

  1128→function formatScheduleDesc(task: ScheduledTask): string {
  1129→  if (task.scheduleType === "cron" && task.cronExpr) {
  1130→    return `cron: ${task.cronExpr} (${task.timezone})`;
  1131→  }
  1132→  if (task.scheduleType === "fixed_rate" && task.intervalSec) {
  1133→    const mins = task.intervalSec / 60;
  1134→    if (mins < 60) return `every ${mins} minute${mins !== 1 ? "s" : ""}`;
  1135→    const hrs = mins / 60;
  1136→    if (hrs < 24) return `every ${hrs} hour${hrs !== 1 ? "s" : ""}`;
  1137→    return `every ${(hrs / 24).toFixed(1)} days`;
  1138→  }
  1139→  if (task.scheduleType === "one_time" && task.runAt) {
  1140→    return `once at ${new Date(task.runAt).toLocaleString()} (${task.timezone})`;
  1141→  }
  1142→  return task.scheduleType;
  1143→}

  1145→function formatRelativeTime(isoStr: string): string {
  1146→  const diff = Date.now() - new Date(isoStr).getTime();
  1147→  const mins = Math.floor(diff / 60000);
  1148→  if (mins < 1) return "just now";
  1149→  if (mins < 60) return `${mins}m ago`;
  1150→  const hours = Math.floor(mins / 60);
  1151→  if (hours < 24) return `${hours}h ago`;
  1152→  const days = Math.floor(hours / 24);
  1153→  return `${days}d ago`;
  1154→}
